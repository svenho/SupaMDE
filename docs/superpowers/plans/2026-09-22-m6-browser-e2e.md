# M6 — Browser-E2E-Tests: Implementierungsplan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** SupaMDE bekommt eine zweite Testebene in einem echten Browser (Chromium), die Layout, Scroll-Geometrie und die nativen Drag-&-Drop-/Clipboard-APIs prüft — alles, was jsdom prinzipbedingt nicht kann.

**Architecture:** Zwei Vitest-Projekte in einer Konfiguration. Das bestehende `unit`-Projekt (jsdom, 568 Tests) bleibt unverändert und wird weiterhin von `npm test` gefahren. Ein neues `browser`-Projekt (Chromium via Playwright-Provider) liegt unter `test/browser/` — bewusst außerhalb von `src/`, damit weder die `.d.ts`-Generierung noch die Coverage es erfasst. Voraussetzung ist die Migration auf Vitest 5, weil `@vitest/browser-playwright` erst ab 5.0 existiert.

**Tech Stack:** Vitest 5, `@vitest/browser`, `@vitest/browser-playwright`, Playwright (Chromium), TypeScript, CodeMirror 6

**Spec:** [docs/superpowers/specs/2026-09-22-supamde-m6-browser-e2e-design.md](../specs/2026-09-22-supamde-m6-browser-e2e-design.md)

## Global Constraints

- **Node:** `engines.node` wird auf `^22.13.0 || >=24` gesetzt (Vitest 5 fordert `>=22.12`). Entwicklungsmaschine: Node 22.18.0.
- **Vite:** bleibt bei `^8.1.5` (Vitest 5 fordert `^6.4.0 || ^7.0.0 || ^8.0.0`).
- **Vitest:** `vitest` und `@vitest/coverage-v8` auf `^5.0.1` — beide Pakete **müssen dieselbe Version** tragen, Vitest erzwingt das über seine Peer-Dependencies.
- **Browser:** ausschließlich Chromium. Kein Firefox, kein WebKit.
- **Provider-API:** `playwright()` als **Funktionsaufruf** aus `@vitest/browser-playwright`, nicht der String `'playwright'` (Vitest-3-Form). `headless` gehört auf die `browser`-Ebene, nicht in den Instanz-Eintrag.
- **Bestehende Tests:** Die 568 Tests in 45 Dateien werden **nicht angefasst**. Nach jeder Task müssen sie grün bleiben.
- **Sprache:** Testbeschreibungen und Kommentare auf Deutsch, passend zum Bestand. Code-Identifier bleiben englisch.
- **Installationen:** Vor jeder Paketinstallation und vor dem Chromium-Download ist die Zustimmung des Auftraggebers einzuholen (Task 1 und Task 2 enthalten dafür je einen expliziten Haltepunkt).

Create `test/browser/helpers.ts`:

```ts
import SupaMDE from '../../src/index';
import type { SupaMDEOptions } from '../../src/options';

/**
 * Baut eine echte SupaMDE-Instanz auf einer Textarea am document.body.
 *
 * Bewusst NICHT der jsdom-Helfer aus src/__tests__/helpers.ts: der baut
 * EditorState/EditorView ohne Layout und ist auf die Unit-Ebene zugeschnitten.
 * Hier geht es um die vollständige Fassade samt Container, Toolbar und CSS.
 */
export function mountEditor(options: SupaMDEOptions = {}): {
  editor: SupaMDE;
  cleanup: () => void;
} {
  const textarea = document.createElement('textarea');
  document.body.appendChild(textarea);

  const editor = new SupaMDE({ element: textarea, ...options });

  const cleanup = (): void => {
    // toTextArea() räumt den Editor geordnet zurück — wichtig, weil
    // fullscreenCount in src/ui/fullscreen.ts MODULWEITER Zustand ist. Stirbt
    // eine Instanz im Fullscreen, bliebe der Zähler stehen und
    // document.body.style.overflow gesperrt; der nächste Test liefe auf einem
    // verfälschten Ausgangszustand.
    try {
      editor.setFullScreen(false);
      editor.toTextArea();
    } catch {
      // Schon zurückgebaut — dann ist nichts mehr zu tun.
    }
    textarea.remove();
    document.body.style.overflow = '';
  };

  return { editor, cleanup };
}

/** Ein Dokument, das sicher länger als jeder Testviewport ist. */
export function langesDokument(zeilen = 200): string {
  return Array.from({ length: zeilen }, (_, i) => `Zeile ${i + 1} mit etwas Text.`).join('\n');
}

/**
 * Wartet einen Frame ab. Nötig, wo der Code per requestAnimationFrame
 * aufräumt — etwa der Guard-Reset in src/ui/preview.ts.
 */
export function naechsterFrame(): Promise<void> {
  return new Promise((resolve) => requestAnimationFrame(() => resolve()));
}
```

```ts
import { describe, it, expect, afterEach } from 'vitest';
import { mountEditor, langesDokument } from './helpers';

let cleanup: (() => void) | null = null;
afterEach(() => {
  cleanup?.();
  cleanup = null;
});

/** Der Container, den die Fassade um den Editor baut. */
function containerOf(editor: { codemirror: { dom: HTMLElement } }): HTMLElement {
  const el = editor.codemirror.dom.closest('.supamde-container');
  if (!el) throw new Error('Container .supamde-container nicht gefunden');
  return el as HTMLElement;
}

describe('Fullscreen-Geometrie', () => {
  it('spannt den Container auf Viewport-Größe', () => {
    const mounted = mountEditor();
    cleanup = mounted.cleanup;
    const container = containerOf(mounted.editor);

    const vorher = container.getBoundingClientRect();
    expect(vorher.height).toBeLessThan(window.innerHeight);

    mounted.editor.setFullScreen(true);

    const nachher = container.getBoundingClientRect();
    expect(Math.round(nachher.width)).toBe(window.innerWidth);
    expect(Math.round(nachher.height)).toBe(window.innerHeight);
  });

  it('hält die Höhenkette: langes Dokument scrollt innen, statt den Viewport zu sprengen', () => {
    // Die Regression aus 11e4f9d. Die drei ineinandergreifenden Flex-Regeln in
    // src/ui/fullscreen.css sind nur mit echtem Layout prüfbar — in jsdom sind
    // sämtliche Höhen 0.
    const mounted = mountEditor({ initialValue: langesDokument() });
    cleanup = mounted.cleanup;
    const container = containerOf(mounted.editor);

    mounted.editor.setFullScreen(true);

    const rect = container.getBoundingClientRect();
    expect(Math.round(rect.height)).toBe(window.innerHeight);

    // Der Scroller muss innen überlaufen — genau das ist "scrollt innen".
    const scroller = mounted.editor.codemirror.scrollDOM;
    expect(scroller.scrollHeight).toBeGreaterThan(scroller.clientHeight);

    // Und der Body darf dabei NICHT mitscrollen.
    expect(document.body.scrollHeight).toBeLessThanOrEqual(window.innerHeight + 1);
  });

  it('sperrt den body-Scroll und gibt ihn wieder frei', () => {
    const mounted = mountEditor({ initialValue: langesDokument() });
    cleanup = mounted.cleanup;

    expect(document.body.style.overflow).not.toBe('hidden');
    mounted.editor.setFullScreen(true);
    expect(document.body.style.overflow).toBe('hidden');
    mounted.editor.setFullScreen(false);
    expect(document.body.style.overflow).not.toBe('hidden');
  });
});

describe('Side-by-Side-Aufteilung', () => {
  it('teilt die Editor-Zeile etwa hälftig', () => {
    const mounted = mountEditor({ initialValue: '# Titel\n\nAbsatz.' });
    cleanup = mounted.cleanup;
    const container = containerOf(mounted.editor);

    mounted.editor.setSideBySide(true);

    const zeile = container.querySelector('.supamde-editor-row') as HTMLElement;
    const editorBreite = mounted.editor.codemirror.dom.getBoundingClientRect().width;
    const panel = container.querySelector('.supamde-preview-side') as HTMLElement;
    const panelBreite = panel.getBoundingClientRect().width;

    expect(panelBreite).toBeGreaterThan(0);
    // 50/50 mit Toleranz für Rahmen und Sub-Pixel-Rundung.
    const zeilenBreite = zeile.getBoundingClientRect().width;
    expect(editorBreite).toBeGreaterThan(zeilenBreite * 0.4);
    expect(panelBreite).toBeGreaterThan(zeilenBreite * 0.4);
  });
});
```

```ts
import { describe, it, expect, afterEach } from 'vitest';
import { mountEditor, langesDokument, naechsterFrame } from './helpers';

let cleanup: (() => void) | null = null;
afterEach(() => {
  cleanup?.();
  cleanup = null;
});

function panelOf(editor: { codemirror: { dom: HTMLElement } }): HTMLElement {
  const container = editor.codemirror.dom.closest('.supamde-container');
  const panel = container?.querySelector('.supamde-preview-side');
  if (!panel) throw new Error('Vorschau-Panel nicht gefunden');
  return panel as HTMLElement;
}

/** Scroll-Fortschritt als Anteil, damit unterschiedliche Höhen vergleichbar sind. */
function anteil(el: HTMLElement): number {
  const denom = el.scrollHeight - el.clientHeight;
  return denom > 0 ? el.scrollTop / denom : 0;
}

describe('Scroll-Sync mit echten Höhen', () => {
  it('zieht die Vorschau mit, wenn der Editor scrollt', async () => {
    const mounted = mountEditor({ initialValue: langesDokument() });
    cleanup = mounted.cleanup;
    mounted.editor.setSideBySide(true);

    const scroller = mounted.editor.codemirror.scrollDOM;
    const panel = panelOf(mounted.editor);

    // Echte Höhen — in jsdom wären beide 0 und der Test bedeutungslos.
    expect(scroller.scrollHeight).toBeGreaterThan(scroller.clientHeight);
    expect(panel.scrollHeight).toBeGreaterThan(panel.clientHeight);

    scroller.scrollTop = (scroller.scrollHeight - scroller.clientHeight) * 0.5;
    scroller.dispatchEvent(new Event('scroll'));
    await naechsterFrame();

    expect(anteil(panel)).toBeCloseTo(0.5, 1);
  });

  it('zieht den Editor mit, wenn die Vorschau scrollt', async () => {
    const mounted = mountEditor({ initialValue: langesDokument() });
    cleanup = mounted.cleanup;
    mounted.editor.setSideBySide(true);

    const scroller = mounted.editor.codemirror.scrollDOM;
    const panel = panelOf(mounted.editor);

    panel.scrollTop = (panel.scrollHeight - panel.clientHeight) * 0.25;
    panel.dispatchEvent(new Event('scroll'));
    await naechsterFrame();

    expect(anteil(scroller)).toBeCloseTo(0.25, 1);
  });

  it('schaukelt sich nicht auf und verschluckt den nächsten Nutzer-Scroll nicht', async () => {
    // Der Daseinszweck von scheduleGuardReset in src/ui/preview.ts. In jsdom
    // greift mangels requestAnimationFrame nur der synchrone Fallback-Zweig —
    // der echte rAF-Pfad bleibt dort ungetestet.
    const mounted = mountEditor({ initialValue: langesDokument() });
    cleanup = mounted.cleanup;
    mounted.editor.setSideBySide(true);

    const scroller = mounted.editor.codemirror.scrollDOM;
    const panel = panelOf(mounted.editor);

    scroller.scrollTop = (scroller.scrollHeight - scroller.clientHeight) * 0.5;
    scroller.dispatchEvent(new Event('scroll'));
    await naechsterFrame();

    // Direkt danach ein echter Nutzer-Scroll: er MUSS ankommen.
    scroller.scrollTop = (scroller.scrollHeight - scroller.clientHeight) * 0.8;
    scroller.dispatchEvent(new Event('scroll'));
    await naechsterFrame();

    expect(anteil(panel)).toBeCloseTo(0.8, 1);
  });

  it('bleibt ruhig, wenn das Dokument kürzer als der Viewport ist', async () => {
    // Randfall denom === 0 in src/ui/preview.ts: kein NaN, kein Sprung.
    const mounted = mountEditor({ initialValue: 'Kurz.' });
    cleanup = mounted.cleanup;
    mounted.editor.setSideBySide(true);

    const scroller = mounted.editor.codemirror.scrollDOM;
    const panel = panelOf(mounted.editor);

    scroller.dispatchEvent(new Event('scroll'));
    await naechsterFrame();

    expect(Number.isNaN(panel.scrollTop)).toBe(false);
    expect(panel.scrollTop).toBe(0);
  });
});
```

- Create: `test/browser/upload.test.ts`

**Interfaces:**
- Consumes: `mountEditor` aus Task 3
- Produces: nichts (Endverbraucher)

**Wichtige Abweichung vom Spec (§5.3):** Das Spec nannte `userEvent.upload()` für den Dateiauswahl-Pfad. Das ist **nicht praktikabel**: `openFilePicker()` in `src/features/upload-dom.ts:44-74` erzeugt den Input bei Bedarf, ruft sofort `input.click()` — was im echten Browser den **Systemdialog** öffnet — und entfernt ihn wieder. Ein Locator auf dieses flüchtige Element ist im Test nicht greifbar, und der Systemdialog blockierte den Lauf.

Stattdessen deckt dieser Test die Upload-Orchestrierung über `uploadImages(files)` ab (die öffentliche API-Methode, die auch der Picker am Ende ruft) sowie Drop und Paste über echte `DataTransfer`-/`ClipboardEvent`-Objekte. Damit bleibt der Kern — Platzhalter, Positionsstabilität, Fehlerfall — vollständig abgedeckt; nur der Systemdialog selbst bleibt außen vor, und der gehört ohnehin dem Betriebssystem.

Geprüfte Default-Texte aus `src/features/image-upload.ts:78-86`:
`placeholder: '![Uploading {name}…]()'`, `errorFailed: 'Upload von {name} fehlgeschlagen.'`

- [ ] **Step 1: Tests schreiben**

Create `test/browser/upload.test.ts`:

```ts
import { describe, it, expect, afterEach, vi } from 'vitest';
import { mountEditor } from './helpers';

let cleanup: (() => void) | null = null;
afterEach(() => {
  cleanup?.();
  cleanup = null;
});

/** Eine echte Bilddatei — im Browser, nicht nachgebaut wie in jsdom. */
function bildDatei(name = 'bild.png'): File {
  const bytes = new Uint8Array([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]);
  return new File([bytes], name, { type: 'image/png' });
}

/** Wartet, bis die Bedingung zutrifft — für asynchrone Upload-Auflösung. */
async function bisGilt(pruefung: () => boolean, ms = 2000): Promise<void> {
  const ende = Date.now() + ms;
  while (Date.now() < ende) {
    if (pruefung()) return;
    await new Promise((r) => setTimeout(r, 16));
  }
  throw new Error('Bedingung wurde nicht erfüllt');
}

describe('Bild-Upload über die öffentliche API', () => {
  it('setzt einen Platzhalter und ersetzt ihn durch die Bild-Syntax', async () => {
    let aufloesen!: (url: string) => void;
    const upload = vi.fn(() => new Promise<string>((r) => { aufloesen = r; }));

    const mounted = mountEditor({
      initialValue: '',
      uploadImage: { enabled: true, upload },
    });
    cleanup = mounted.cleanup;

    mounted.editor.uploadImages([bildDatei()]);
    await bisGilt(() => mounted.editor.getValue().includes('Uploading'));
    expect(mounted.editor.getValue()).toContain('![Uploading bild.png…]()');

    aufloesen('https://example.test/bild.png');
    await bisGilt(() => mounted.editor.getValue().includes('example.test'));

    const text = mounted.editor.getValue();
    expect(text).toContain('![bild.png](https://example.test/bild.png)');
    expect(text).not.toContain('Uploading');
  });

  it('hält die Platzhalter-Position, wenn davor getippt wird', async () => {
    // Der Daseinszweck des StateField in src/features/upload-placeholder.ts.
    let aufloesen!: (url: string) => void;
    const upload = vi.fn(() => new Promise<string>((r) => { aufloesen = r; }));

    const mounted = mountEditor({
      initialValue: 'ENDE',
      uploadImage: { enabled: true, upload },
    });
    cleanup = mounted.cleanup;

    // Cursor ans Dokumentende, dort entsteht der Platzhalter.
    const laenge = mounted.editor.codemirror.state.doc.length;
    mounted.editor.codemirror.dispatch({ selection: { anchor: laenge } });
    mounted.editor.uploadImages([bildDatei('x.png')]);
    await bisGilt(() => mounted.editor.getValue().includes('Uploading'));

    // Jetzt VOR dem Platzhalter einfügen — die Positionen verschieben sich.
    mounted.editor.codemirror.dispatch({ changes: { from: 0, insert: 'NEU ' } });

    aufloesen('https://example.test/x.png');
    await bisGilt(() => mounted.editor.getValue().includes('example.test'));

    const text = mounted.editor.getValue();
    expect(text).toContain('NEU ENDE');
    expect(text).toContain('![x.png](https://example.test/x.png)');
    expect(text).not.toContain('Uploading');
  });

  it('ersetzt den Platzhalter bei Fehlschlag durch den Fehlertext', async () => {
    const upload = vi.fn(() => Promise.reject(new Error('Netzwerk')));
    const onError = vi.fn();

    const mounted = mountEditor({
      initialValue: '',
      uploadImage: { enabled: true, upload, onError },
    });
    cleanup = mounted.cleanup;

    mounted.editor.uploadImages([bildDatei('kaputt.png')]);
    await bisGilt(() => !mounted.editor.getValue().includes('Uploading'));

    expect(onError).toHaveBeenCalledTimes(1);
    expect(onError.mock.calls[0][0].kind).toBe('upload-failed');
    expect(mounted.editor.getValue()).not.toContain('Uploading');
  });
});

describe('Drop und Paste mit echten Browser-Objekten', () => {
  it('lädt eine per Drop übergebene Bilddatei hoch', async () => {
    const upload = vi.fn(() => Promise.resolve('https://example.test/drop.png'));
    const mounted = mountEditor({
      initialValue: '',
      uploadImage: { enabled: true, upload },
    });
    cleanup = mounted.cleanup;

    // Echter DataTransfer — in jsdom ist `files` nicht befüllbar und musste
    // per Objektliteral nachgebaut werden (siehe upload-dom.test.ts:13).
    const dt = new DataTransfer();
    dt.items.add(bildDatei('drop.png'));
    expect(dt.files.length).toBe(1);

    mounted.editor.codemirror.contentDOM.dispatchEvent(
      new DragEvent('drop', { dataTransfer: dt, bubbles: true, cancelable: true }),
    );

    await bisGilt(() => mounted.editor.getValue().includes('example.test'));
    expect(mounted.editor.getValue()).toContain('![drop.png](https://example.test/drop.png)');
  });

  it('lädt eine per Paste übergebene Bilddatei hoch', async () => {
    const upload = vi.fn(() => Promise.resolve('https://example.test/paste.png'));
    const mounted = mountEditor({
      initialValue: '',
      uploadImage: { enabled: true, upload },
    });
    cleanup = mounted.cleanup;

    const dt = new DataTransfer();
    dt.items.add(bildDatei('paste.png'));

    mounted.editor.codemirror.contentDOM.dispatchEvent(
      new ClipboardEvent('paste', { clipboardData: dt, bubbles: true, cancelable: true }),
    );

    await bisGilt(() => mounted.editor.getValue().includes('example.test'));
    expect(mounted.editor.getValue()).toContain('![paste.png](https://example.test/paste.png)');
  });
});
```
